function submeterRespostaMapa_(dados) {
  dados = dados || {};

  var casoId = String(dados.casoId || '').trim();
  var respondenteNum = Number(dados.respondenteNum);
  var respostas = dados.respostas || {};

  if (!casoId) {
    return { status: 'erro', mensagem: 'Caso inválido.' };
  }

  if (![1, 2, 3].includes(respondenteNum)) {
    return { status: 'erro', mensagem: 'Respondente inválido.' };
  }

  // Mantenha aqui a validação já existente que confirma que casoId pertence
  // a um caso Mapa válido/ativo e a um Pedido Mapa Pago.
  // NÃO remova essa validação do código atual ao fazer a substituição.

  var codigosObrigatorios = [
    'D1.3','D1.1','D2.1','D2.3','D3.1','D3.2',
    'D4.1','D4.2','D5.1','D5.2','D6.1','D6.2'
  ];
  var notasPermitidas = [0, 5, 10];

  for (var i = 0; i < codigosObrigatorios.length; i++) {
    var codigo = codigosObrigatorios[i];
    var item = respostas[codigo];

    if (!item || notasPermitidas.indexOf(Number(item.nota)) === -1) {
      return {
        status: 'erro',
        mensagem: 'As 12 respostas são obrigatórias e devem conter uma opção válida.'
      };
    }
  }

  var respNome = String(dados.respNome || '').trim();
  var respFuncao = String(dados.respFuncao || '').trim();
  var respEmpresa = String(dados.respEmpresa || '').trim();
  var respEmail = String(dados.respEmail || '').trim();
  var respWhatsapp = String(dados.respWhatsapp || '').trim();

  if (!respNome || !respFuncao || !respEmpresa || !respEmail || !respWhatsapp) {
    return { status: 'erro', mensagem: 'Identificação do respondente incompleta.' };
  }

  var ss = SpreadsheetApp.openById(PLANILHA_ID);
  var sheet = ss.getSheetByName('Mapa_Respostas');
  if (!sheet) {
    return { status: 'erro', mensagem: 'Estrutura de respostas indisponível.' };
  }

  // Idempotência: um mesmo slot de respondente só pode concluir uma vez.
  // Pressupõe Caso ID na coluna A e Respondente Num na coluna B.
  // Se sua aba usa colunas diferentes, ajuste SOMENTE estes dois índices.
  var ultimaLinha = sheet.getLastRow();
  if (ultimaLinha >= 2) {
    var existentes = sheet.getRange(2, 1, ultimaLinha - 1, 2).getValues();
    for (var j = 0; j < existentes.length; j++) {
      if (
        String(existentes[j][0]).trim() === casoId &&
        Number(existentes[j][1]) === respondenteNum
      ) {
        return {
          status: 'erro',
          mensagem: 'Este link de respondente já foi concluído.'
        };
      }
    }
  }

  /*
   * A PARTIR DAQUI:
   * mantenha exatamente a montagem da variável "linha" existente na função atual,
   * porque ela contém a ordem real das colunas da sua aba Mapa_Respostas.
   *
   * Depois mantenha:
   *
   * sheet.appendRow(linha);
   * return { status: 'ok' };
   *
   * Não invente/reordene colunas.
   */
}
